## 🧱 Step 3. Build a U-Net model in Keras

Here’s a simple but solid 2D U-Net implementation compatible with your data:

```python
import tensorflow as tf
from tensorflow.keras import layers, Model

def unet_model(input_shape, num_classes):
    inputs = layers.Input(shape=input_shape)

    # Encoder
    c1 = layers.Conv2D(32, (3, 3), activation='relu', padding='same')(inputs)
    c1 = layers.Conv2D(32, (3, 3), activation='relu', padding='same')(c1)
    p1 = layers.MaxPooling2D((2, 2))(c1)

    c2 = layers.Conv2D(64, (3, 3), activation='relu', padding='same')(p1)
    c2 = layers.Conv2D(64, (3, 3), activation='relu', padding='same')(c2)
    p2 = layers.MaxPooling2D((2, 2))(c2)

    c3 = layers.Conv2D(128, (3, 3), activation='relu', padding='same')(p2)
    c3 = layers.Conv2D(128, (3, 3), activation='relu', padding='same')(c3)
    p3 = layers.MaxPooling2D((2, 2))(c3)

    # Bottleneck
    b = layers.Conv2D(256, (3, 3), activation='relu', padding='same')(p3)
    b = layers.Conv2D(256, (3, 3), activation='relu', padding='same')(b)

    # Decoder
    u3 = layers.UpSampling2D((2, 2))(b)
    u3 = layers.concatenate([u3, c3])
    c4 = layers.Conv2D(128, (3, 3), activation='relu', padding='same')(u3)
    c4 = layers.Conv2D(128, (3, 3), activation='relu', padding='same')(c4)

    u2 = layers.UpSampling2D((2, 2))(c4)
    u2 = layers.concatenate([u2, c2])
    c5 = layers.Conv2D(64, (3, 3), activation='relu', padding='same')(u2)
    c5 = layers.Conv2D(64, (3, 3), activation='relu', padding='same')(c5)

    u1 = layers.UpSampling2D((2, 2))(c5)
    u1 = layers.concatenate([u1, c1])
    c6 = layers.Conv2D(32, (3, 3), activation='relu', padding='same')(u1)
    c6 = layers.Conv2D(32, (3, 3), activation='relu', padding='same')(c6)

    # Output
    outputs = layers.Conv2D(num_classes, (1, 1), activation='softmax')(c6)

    model = Model(inputs, outputs)
    return model
```

```python
import numpy as np
import nibabel as nib
import tensorflow as tf
from tensorflow.keras.utils import Sequence
from random import shuffle

class DataGenerator2D(Sequence):
    def __init__(self, image_files, mask_files, batch_size=8,
                 normImage=True, categorical=True, dtype=np.float32, shuffle_data=True):
        self.image_files = image_files
        self.mask_files = mask_files
        self.batch_size = batch_size
        self.normImage = normImage
        self.categorical = categorical
        self.dtype = dtype
        self.shuffle_data = shuffle_data
        self.on_epoch_end()

    def __len__(self):
        return int(np.ceil(len(self.image_files) / self.batch_size))

    def on_epoch_end(self):
        self.indices = np.arange(len(self.image_files))
        if self.shuffle_data:
            np.random.shuffle(self.indices)

    def __getitem__(self, idx):
        batch_indices = self.indices[idx*self.batch_size:(idx+1)*self.batch_size]
        X_batch, y_batch = self._load_batch(batch_indices)
        return X_batch, y_batch

    def _load_batch(self, batch_indices):
        batch_imgs = []
        batch_masks = []

        for i in batch_indices:
            # Load image
            img = nib.load(self.image_files[i]).get_fdata(caching='unchanged')
            if img.ndim == 3:
                img = img[:, :, 0]  # remove extra dim if present
            img = img.astype(self.dtype)
            if self.normImage:
                img = (img - img.mean()) / (img.std() + 1e-8)

            # Load mask
            mask = nib.load(self.mask_files[i]).get_fdata(caching='unchanged')
            if mask.ndim == 3:
                mask = mask[:, :, 0]
            mask = mask.astype(np.uint8)

            if self.categorical:
                mask = self._to_channels(mask)

            batch_imgs.append(img)
            batch_masks.append(mask)

        X = np.stack(batch_imgs, axis=0)
        y = np.stack(batch_masks, axis=0)

        # Add channel dim if grayscale
        if X.ndim == 3:
            X = X[..., np.newaxis]

        return X, y

    def _to_channels(self, arr):
        """Convert mask to one-hot channel representation"""
        channels = np.unique(arr)
        res = np.zeros(arr.shape + (len(channels),), dtype=self.dtype)
        for c in channels:
            c = int(c)
            res[..., c:c+1][arr == c] = 1
        return res
```

Excellent question — this is subtle but important Python behavior to understand. 👇

Let’s unpack it carefully.

---

## 🧠 What happens when you slice past the end of a list in Python

If you do:

```python
my_list = [10, 20, 30]
start_index = 5
subset = my_list[start_index:]
```

Python **does not throw an error**.
Instead, you simply get an **empty list**:

```python
>>> subset
[]
```

That’s because Python slicing is *graceful*:

* Going **past the end** returns an empty list.
* Going **partially past** just gives you up to the last element.
* Using negative indices works from the end (e.g., `[-2:]`).

So in your case:

```python
self.images = self.images[start_index:]
```

If `start_index >= len(self.images)`,
→ `self.images` becomes an **empty list** `[]`.

Same goes for `self.masks`.

---

## 🧨 Why that can be a problem

While Python doesn’t crash, your training pipeline *will* — later, when you try to load batches or train your model.
For example:

* Your `__getitem__` might try to index into an empty list.
* The model might receive no samples and fail during training.

So it’s **safer to check and raise an error** (or print a clear warning) if this happens.

---

## ✅ Safe version

Here’s the guard you should use:

```python
if start_index >= len(self.images):
    raise ValueError(
        f"start_index ({start_index}) is beyond available dataset size ({len(self.images)})."
    )
```

If you want to be more forgiving (e.g., just warn), you could instead do:

```python
if start_index >= len(self.images):
    print(f"⚠️ start_index ({start_index}) beyond dataset size ({len(self.images)}). No data will be loaded.")
    self.images, self.masks = [], []
```

But in most cases, raising an error is better — because an empty dataset usually means a path or indexing mistake.

---

### ✅ TL;DR

| Case                       | Result                                 |
| -------------------------- | -------------------------------------- |
